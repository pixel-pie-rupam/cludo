        // Initial page load simulation
        let progress = 0;
        const loadingMessages = [
            'Initializing campus map',
            'Loading challenge data',
            'Syncing QR locations',
            'Preparing hunt interface',
            'Finalizing setup',
            'Ready to begin'
        ];

        const initialLoadInterval = setInterval(() => {
            progress += Math.random() * 12 + 3;
            if (progress > 100) progress = 100;
            
            document.getElementById('initialProgress').style.width = progress + '%';
            
            const msgIndex = Math.min(Math.floor((progress / 100) * loadingMessages.length), loadingMessages.length - 1);
            document.getElementById('initialLoadText').textContent = loadingMessages[msgIndex];
            
            if (progress >= 100) {
                clearInterval(initialLoadInterval);
                setTimeout(() => {
                    document.getElementById('initialLoading').classList.add('fade-out');
                    setTimeout(() => {
                        document.getElementById('eventPanel').classList.add('show');
                    }, 800);
                }, 600);
            }
        }, 150);

        // Start adventure button handler
        function startAdventure() {
            // Haptic feedback for mobile (if supported)
            if (navigator.vibrate) {
                navigator.vibrate(50);
            }

            // Hide event panel
            document.getElementById('eventPanel').classList.remove('show');
            
            // Show navigation loading
            document.getElementById('navLoading').classList.add('show');
            
            // Simulate loading files from server
            let navProgress = 0;
            const navMessages = [
                'Loading hunt resources',
                'Connecting to server',
                'Fetching challenge database',
                'Loading map assets',
                'Preparing authentication',
                'Almost there'
            ];
            
            const navLoadInterval = setInterval(() => {
                navProgress += Math.random() * 10 + 5;
                if (navProgress > 100) navProgress = 100;
                
                document.getElementById('navProgress').style.width = navProgress + '%';
                
                const msgIndex = Math.min(Math.floor((navProgress / 100) * navMessages.length), navMessages.length - 1);
                document.getElementById('navLoadText').textContent = navMessages[msgIndex];
                
                if (navProgress >= 100) {
                    clearInterval(navLoadInterval);
                    setTimeout(() => {
                        // Replace with your actual page URL
                        window.location.href = 'trasure-hunt.html';
                    }, 500);
                }
            }, 200);
        }

        // Prevent pull-to-refresh on mobile
        document.body.addEventListener('touchmove', function(e) {
            if (e.target.closest('.event-panel') && document.getElementById('eventPanel').classList.contains('show')) {
                // Allow scrolling in event panel
                return;
            }
        }, { passive: false });
    </script>

            db.collection('leaderboard_config').doc('settings').onSnapshot((doc) => {
                if (doc.exists) {
                    const data = doc.data();
                    const newVisibility = data.isVisible || false;
                    
                    if (newVisibility !== isVisible) {
                        loadLeaderboard();
                    }
                }
            });

            db.collection('teams').onSnapshot(() => {
                if (isVisible) {
                    loadLeaderboard();
                }
            });
        }

        // Initialize
        async function initialize() {
            console.log('Initializing leaderboard...');
            
            hideElement(elements.errorState);
            showElement(elements.loadingState);
            
            try {
                await loadLeaderboard();
                setupRealtimeListener();
                
                // Auto-refresh every 30 seconds
                setInterval(() => {
                    if (isVisible && !document.hidden) {
                        loadLeaderboard();
                    }
                }, 30000);

            } catch (err) {
                console.error('Initialization error:', err);
                showError('Failed to initialize. Please refresh the page.');
            }
        }

        // Prevent back navigation on mobile
        history.pushState(null, null, location.href);
        window.onpopstate = function () {
            history.go(1);
        };

        // Start the app
        initialize();
    </script>

        // Location Tracking Functions
        async function requestLocationPermission(teamId, teamName) {
            if (!navigator.geolocation) {
                alert('Geolocation is not supported by your browser');
                return;
            }

            // Check if Permissions API is available (not supported in all mobile browsers)
            if (navigator.permissions && navigator.permissions.query) {
                try {
                    const result = await navigator.permissions.query({ name: 'geolocation' });
                    
                    if (result.state === 'granted') {
                        // Permission already granted, start tracking directly
                        startLocationTracking(teamId, teamName);
                        locationStatusBanner.classList.remove('hidden');
                        return;
                    } else if (result.state === 'denied') {
                        // Permission was previously denied, show instructions
                        showPermissionBlockedInstructions();
                    }
                } catch (error) {
                    console.log('Permissions API not fully supported, will request directly');
                }
            }

            locationPermissionModal.classList.remove('hidden');
            setupLocationPermissionListeners(teamId, teamName);
        }

        function setupLocationPermissionListeners(teamId, teamName) {
            const requestPermission = async () => {
                // Disable button and show loading state
                grantLocationBtn.disabled = true;
                grantLocationBtn.innerHTML = '<div class="spinner mx-auto"></div>';
                
                try {
                    // Request location permission - this should trigger browser's permission dialog
                    const position = await new Promise((resolve, reject) => {
                        navigator.geolocation.getCurrentPosition(
                            resolve,
                            reject,
                            { 
                                enableHighAccuracy: true, 
                                timeout: 15000, 
                                maximumAge: 0 
                            }
                        );
                    });
                    
                    // Permission granted successfully
                    locationPermissionModal.classList.add('hidden');
                    startLocationTracking(teamId, teamName);
                    locationStatusBanner.classList.remove('hidden');
                    
                } catch (error) {
                    // Permission denied or error occurred
                    console.error('Location error:', error);
                    handleLocationError(error);
                }
            };

            // Primary button handler
            grantLocationBtn.addEventListener('click', requestPermission, { once: true });
            
            // Retry button handler
            retryLocationBtn.addEventListener('click', () => {
                // Hide instructions and retry button
                permissionInstructions.classList.add('hidden');
                retryLocationBtn.classList.add('hidden');
                grantLocationBtn.classList.remove('hidden');
                
                // Setup new listener for retry
                grantLocationBtn.addEventListener('click', requestPermission, { once: true });
            });
        }

        function handleLocationError(error) {
            grantLocationBtn.disabled = false;
            
            let errorMessage = '';
            
            if (error.code === 1) { // PERMISSION_DENIED
                errorMessage = 'Location access was denied. Please check the instructions below to enable it.';
                showPermissionBlockedInstructions();
            } else if (error.code === 2) { // POSITION_UNAVAILABLE
                errorMessage = 'Location information is unavailable. Please ensure GPS is enabled on your device.';
                grantLocationBtn.innerHTML = '<i class="fas fa-check-circle mr-2"></i>Grant Location Access';
                alert(errorMessage);
            } else if (error.code === 3) { // TIMEOUT
                errorMessage = 'Location request timed out. Please check your connection and try again.';
                grantLocationBtn.innerHTML = '<i class="fas fa-check-circle mr-2"></i>Grant Location Access';
                alert(errorMessage);
            } else {
                errorMessage = 'An unknown error occurred. Please try again.';
                grantLocationBtn.innerHTML = '<i class="fas fa-check-circle mr-2"></i>Grant Location Access';
                alert(errorMessage);
            }
        }

        function startLocationTracking(teamId, teamName) {
            appState.locationWatchId = navigator.geolocation.watchPosition(
                (position) => {
                    appState.currentLocation = {
                        lat: position.coords.latitude,
                        lng: position.coords.longitude,
                        accuracy: position.coords.accuracy,
                        timestamp: new Date()
                    };
                    
                    appState.locationHistory.push({...appState.currentLocation});
                    if (appState.locationHistory.length > MAX_HISTORY_SIZE) {
                        appState.locationHistory.shift();
                    }
                },
                (error) => {
                    console.error('Location tracking error:', error);
                    document.getElementById('location-status-text').textContent = 'Location tracking error';
                },
                { enableHighAccuracy: true, timeout: 20000, maximumAge: 0 }
            );

            appState.locationUpdateInterval = setInterval(() => {
                if (appState.currentLocation) {
                    updateLocationInFirebase(teamId, teamName, appState.currentLocation);
                }
            }, 20000);

            if (appState.currentLocation) {
                updateLocationInFirebase(teamId, teamName, appState.currentLocation);
            }
        }

        async function updateLocationInFirebase(teamId, teamName, location) {
            try {
                const locationData = {
                    teamId: teamId,
                    teamName: teamName,
                    currentLocation: {
                        lat: location.lat,
                        lng: location.lng,
                        accuracy: location.accuracy
                    },
                    lastUpdated: firebase.firestore.FieldValue.serverTimestamp(),
                    isActive: true,
                    recentHistory: appState.locationHistory.slice(-10).map(loc => ({
                        lat: loc.lat,
                        lng: loc.lng,
                        timestamp: loc.timestamp
                    }))
                };

                await db.collection('team_locations').doc(teamId).set(locationData, { merge: true });

                await db.collection('team_locations').doc(teamId)
                    .collection('history').add({
                        lat: location.lat,
                        lng: location.lng,
                        accuracy: location.accuracy,
                        timestamp: firebase.firestore.FieldValue.serverTimestamp()
                    });

                console.log('Location updated successfully');
            } catch (error) {
                console.error('Error updating location:', error);
            }
        }

        async function handleHintRequest(clue) {
            if (appState.currentTeam.score < 30) { return alert("You don't have enough points for a hint!"); }
            if (!confirm("Are you sure you want to spend 30 points for a hint?")) return;
            hintButton.disabled = true;
            hintButton.innerHTML = '<div class="spinner mx-auto"></div>';
            showLoader(true);
            try {
                const teamRef = db.collection('teams').doc(appState.currentTeam.id);
                await teamRef.update({
                    score: firebase.firestore.FieldValue.increment(-30),
                    hintsUsed: firebase.firestore.FieldValue.increment(1)
                });
                hintText.innerHTML = `<div class="flex items-start space-x-3"><i class="fas fa-lightbulb text-yellow-400 mt-1"></i><div>${clue.hint}</div></div>`;
                hintText.classList.remove('hidden');
            } catch (error) {
                alert("Could not process hint request. Please try again.");
                hintButton.disabled = false;
                hintButton.innerHTML = '<i class="fas fa-lightbulb mr-2"></i>Get Hint (-30 Points)';
            } finally {
                showLoader(false);
            }
        }

        async function handleFlagSubmission(clue) {
            const submittedFlag = modalFlagInput.value.trim();
            modalSubmitButton.disabled = true;
            modalSubmitButton.innerHTML = `<div class="spinner mr-2"></div>Submitting...`;

            const teamRef = db.collection('teams').doc(appState.currentTeam.id);
            const clueRef = db.collection('clues').doc(clue.id);
            let pointsToAward = 0;
            let isRabbitHole = false;

            if (submittedFlag === clue.flag) {
                const solverCount = Object.keys(clue.solvers || {}).length;
                if (solverCount === 0) pointsToAward = 250;
                else if (solverCount <= 3) pointsToAward = 200;
                else pointsToAward = 150;
            } else if (clue.rabbitHoleFlag && submittedFlag === clue.rabbitHoleFlag) {
                pointsToAward = -100;
                isRabbitHole = true;
            } else {
                modalFeedback.innerHTML = '<i class="fas fa-times-circle mr-2"></i>Incorrect Flag.';
                modalFeedback.className = 'text-center mt-4 h-4 font-medium text-red-400 text-sm';
                modalSubmitButton.disabled = false;
                modalSubmitButton.innerHTML = '<i class="fas fa-paper-plane mr-2"></i>Submit Flag';
                return;
            }

            try {
                await db.runTransaction(async t => {
                    const clueDoc = await t.get(clueRef);
                    if (!clueDoc.exists) throw new Error("This clue has disappeared!");
                    const clueData = clueDoc.data();

                    if (!isRabbitHole && (clueData.solvers || {})[appState.currentTeam.id]) {
                        throw new Error("You have already solved this clue!");
                    }
                    
                    const teamUpdate = {
                        score: firebase.firestore.FieldValue.increment(pointsToAward),
                        lastSubmissionTimestamp: firebase.firestore.FieldValue.serverTimestamp()
                    };

                    if (isRabbitHole) {
                        const rabbitHoleId = `rabbit_${Date.now()}`;
                        teamUpdate[`rabbitHoles.${rabbitHoleId}`] = {
                            clueTitle: clue.clueTitle,
                            timestamp: firebase.firestore.FieldValue.serverTimestamp()
                        };
                    } else {
                        teamUpdate.solvedClues = firebase.firestore.FieldValue.arrayUnion(clue.id);
                    }
                    
                    t.update(teamRef, teamUpdate);

                    if (!isRabbitHole) {
                        const newSolvers = { ...(clueData.solvers || {}) };
                        newSolvers[appState.currentTeam.id] = { teamName: appState.currentTeam.teamName, solvedAt: new Date() };
                        t.update(clueRef, { solvers: newSolvers });
                    }
                });

                const message = isRabbitHole 
                    ? `Team "${appState.currentTeam.teamName}" fell into a rabbit hole!` 
                    : `Team "${appState.currentTeam.teamName}" solved '${clue.clueTitle}' for ${pointsToAward} points!`;
                await db.collection('event_log').add({ message, timestamp: new Date() });

                modalFeedback.innerHTML = isRabbitHole 
                    ? `<i class="fas fa-exclamation-triangle mr-2"></i>It's a trap! You lose 100 points.`
                    : `<i class="fas fa-check-circle mr-2"></i>Success! You earned ${pointsToAward} points!`;
                modalFeedback.className = `text-center mt-4 h-4 font-medium ${isRabbitHole ? 'text-orange-400' : 'text-green-400'} text-sm`;
                
                setTimeout(() => clueModal.classList.add('hidden'), 2000);

            } catch (error) {
                modalFeedback.innerHTML = `<i class="fas fa-exclamation-triangle mr-2"></i>${error.message}`;
                modalFeedback.className = 'text-center mt-4 h-4 font-medium text-red-400 text-sm';
            } finally {
                modalSubmitButton.disabled = false;
                modalSubmitButton.innerHTML = '<i class="fas fa-paper-plane mr-2"></i>Submit Flag';
            }
        }

                alert('Unable to access camera. Please ensure camera permissions are granted and try again.');
            }
        });

        stopScannerButton.addEventListener('click', () => {
            qrScannerManager.stopScanner();
        });

        qrScannerModal.addEventListener('click', (e) => {
            if (e.target === qrScannerModal) {
                qrScannerManager.stopScanner();
                qrScannerModal.classList.add('hidden');
            }
        });

        // Event Listeners
        closeBroadcast.addEventListener('click', hideBroadcast);
        loginPasswordInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') loginButton.click(); });
        modalFlagInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') modalSubmitButton.click(); });

        clueModal.addEventListener('click', (e) => {
            if (e.target === clueModal) {
                clueModal.classList.add('hidden');
            }
        });

        window.addEventListener('orientationchange', () => {
            setTimeout(() => {
                window.scrollTo(0, 0);
                if (qrScannerManager.isActive) {
                    qrScannerManager.stopScanner();
                    setTimeout(() => {
                        if (qrScannerModal && !qrScannerModal.classList.contains('hidden')) {
                            qrScannerManager.startScanner();
                        }
                    }, 1000);
                }
            }, 500);
        });

        window.addEventListener('beforeunload', () => {
            qrScannerManager.stopScanner();
            stopLocationTracking();
        });

        document.addEventListener('visibilitychange', () => {
            if (document.hidden && qrScannerManager.isActive) {
                qrScannerManager.stopScanner();
            }
        });
    </script>
